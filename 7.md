## Write an SQL query to find the winner in each group.
- The winner in each group is the player who scored the maximum total points within the group. In the case of a tie,
- The lowest player_id wins.

```sql
CREATE TABLE players (
    player_id INT PRIMARY KEY,
    group_id  INT
);

CREATE TABLE matches (
    match_id      INT PRIMARY KEY,
    first_player  INT,
    second_player INT,
    first_score   INT,
    second_score  INT
);
```

```sql
WITH PlayerScores AS (
    SELECT 
        p.group_id,
        p.player_id,
        SUM(CASE 
            WHEN m.first_player = p.player_id THEN m.first_score 
            WHEN m.second_player = p.player_id THEN m.second_score 
            ELSE 0 
        END) AS total_points
    FROM players p
    LEFT JOIN matches m ON p.player_id = m.first_player OR p.player_id = m.second_player
    GROUP BY p.group_id, p.player_id
),
RankedPlayers AS (
    SELECT 
        group_id,
        player_id,
        RANK() OVER (
            PARTITION BY group_id 
            ORDER BY total_points DESC, player_id ASC
        ) AS winner_rank
    FROM PlayerScores
)
SELECT group_id, player_id AS winner_id
FROM RankedPlayers
WHERE winner_rank = 1;
```
```sql
WITH player_scores AS (
    -- Combine scores from both first_player and second_player columns into one list
    SELECT first_player AS player_id, first_score AS score FROM matches
    UNION ALL
    SELECT second_player AS player_id, second_score AS score FROM matches
),
final_scores AS (
    -- Join with players table to get group_id and sum points per player
    SELECT 
        p.group_id,
        ps.player_id,
        SUM(ps.score) AS score
    FROM player_scores ps
    INNER JOIN players p ON p.player_id = ps.player_id
    GROUP BY p.group_id, ps.player_id
),
final_ranking AS (
    -- Rank players within each group: High score first, then lowest ID for ties
    SELECT 
        *,
        RANK() OVER(PARTITION BY group_id ORDER BY score DESC, player_id ASC) AS rn
    FROM final_scores
)
-- Select only the top-ranked player (the winner) from each group
SELECT 
    group_id, 
    player_id AS winner_id
FROM final_ranking
WHERE rn = 1;
```
