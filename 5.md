```sql
CREATE TABLE Friends (
    PersonID INT,
    FriendID INT,
    -- Ensuring the pair is unique and referencing the main table
    PRIMARY KEY (PersonID, FriendID),
    FOREIGN KEY (PersonID) REFERENCES Persons(PersonID),
    FOREIGN KEY (FriendID) REFERENCES Persons(PersonID)
);
```

```sql
SELECT 
    p.PersonID,
    p.Name,
    COUNT(f.FriendID) AS total_friends,
    SUM(f_info.Score) AS sum_of_friend_marks
FROM Persons p
JOIN Friends f 
    ON p.PersonID = f.PersonID
JOIN Persons f_info 
    ON f.FriendID = f_info.PersonID
GROUP BY p.PersonID, p.Name
HAVING SUM(f_info.Score) > 100
ORDER BY p.PersonID;
```
