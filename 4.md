```sql
SET @today_date = '2022-01-02'; -- Note: In 2022, Jan 2nd was actually a Sunday
SET @n = 3;

SELECT 
    DATE_ADD(
        DATE_ADD(@today_date, INTERVAL (8 - DAYOFWEEK(@today_date)) DAY), 
        INTERVAL (@n - 1) WEEK
    ) AS calculated_date;
```
## Output :

| calculated_date |
|:--|
| 2022-01-23      |
