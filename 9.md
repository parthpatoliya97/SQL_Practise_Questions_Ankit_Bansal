```sql
WITH cte AS (
    SELECT 
        date_value,
        state,
        ROW_NUMBER() OVER(PARTITION BY state ORDER BY date_value) AS rn,
        DATE_SUB(date_value, INTERVAL ROW_NUMBER() OVER(PARTITION BY state ORDER BY date_value) DAY) AS common_date
    FROM tasks
)
SELECT 
    MIN(date_value) AS start_date,
    MAX(date_value) AS end_date,
    state
FROM cte
GROUP BY 
    common_date, 
    state
ORDER BY 
    start_date;
```
