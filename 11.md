```sql
CREATE TABLE tournament_results (
    year INT PRIMARY KEY,
    Wimbledon INT,
    Fr_open INT,
    US_open INT,
    Au_open INT
);
```

```sql
SELECT 
    p.player_id, 
    p.player_name,
    SUM(
        (CASE WHEN tr.Wimbledon = p.player_id THEN 1 ELSE 0 END) +
        (CASE WHEN tr.Fr_open = p.player_id THEN 1 ELSE 0 END) +
        (CASE WHEN tr.US_open = p.player_id THEN 1 ELSE 0 END) +
        (CASE WHEN tr.Au_open = p.player_id THEN 1 ELSE 0 END)
    ) AS grand_slams_count
FROM players p
JOIN tournament_results tr 
    ON p.player_id IN (tr.Wimbledon, tr.Fr_open, tr.US_open, tr.Au_open)
GROUP BY p.player_id, p.player_name
ORDER BY grand_slams_count DESC;
```
