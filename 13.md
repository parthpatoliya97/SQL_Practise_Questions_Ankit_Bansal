```sql
WITH company_totals AS (
    SELECT 
        SUM(salary) AS total_salary,
        COUNT(*) AS total_employees
    FROM employee
),

employee_with_totals AS (
    SELECT 
        e.id,
        e.salary,
        e.department,
        c.total_salary,
        c.total_employees
    FROM employee e 
    CROSS JOIN company_totals c
),

excluded_metrics AS (
    SELECT 
        *,
        (total_salary - salary) AS excluded_salary,
        (total_employees - 1) AS excluded_employee -- Fixed: Removed trailing comma here
    FROM employee_with_totals
),

calculated_averages AS (
    SELECT 
        *,
        ROUND(excluded_salary * 1.0 / excluded_employee, 2) AS average_salary
    FROM excluded_metrics
)

SELECT 
    id,
    salary,
    department,
    average_salary
FROM calculated_averages
WHERE salary > average_salary;
```
